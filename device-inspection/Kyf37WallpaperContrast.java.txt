package org.kyf37.home;

import android.app.WallpaperColors;
import android.app.WallpaperManager;
import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.content.IntentFilter;
import android.graphics.Bitmap;
import android.graphics.Canvas;
import android.graphics.drawable.Drawable;
import android.os.Handler;
import android.util.Log;
import android.widget.TextView;

/** Small wallpaper sample; no continuous polling or full-resolution copy retained. */
public final class WallpaperContrast {
    private final Context context;
    private final WallpaperManager manager;
    private final Runnable changed;
    private Bitmap sample;
    private int fallback=ContrastMath.LIGHT;
    private int screenWidth,screenHeight;
    private boolean started;
    private final BroadcastReceiver receiver=new BroadcastReceiver(){
        @Override public void onReceive(Context c,Intent i){refresh();changed.run();}
    };
    private final WallpaperManager.OnColorsChangedListener listener=new WallpaperManager.OnColorsChangedListener(){
        @Override public void onColorsChanged(WallpaperColors colors,int which){
            if((which&WallpaperManager.FLAG_SYSTEM)!=0){refresh();changed.run();}
        }
    };
    public WallpaperContrast(Context c,Runnable update){context=c;manager=WallpaperManager.getInstance(c);changed=update;}
    public void start(){
        if(started)return;
        context.registerReceiver(receiver,new IntentFilter(Intent.ACTION_WALLPAPER_CHANGED));
        manager.addOnColorsChangedListener(listener,new Handler());started=true;
    }
    public void stop(){
        if(!started)return;
        context.unregisterReceiver(receiver);manager.removeOnColorsChangedListener(listener);started=false;
    }
    public void refresh(){
        sample=null;
        // Text coordinates include system bars, so sample against the full display too.
        android.util.DisplayMetrics metrics=new android.util.DisplayMetrics();
        ((android.view.WindowManager)context.getSystemService(Context.WINDOW_SERVICE)).getDefaultDisplay().getRealMetrics(metrics);
        screenWidth=metrics.widthPixels;screenHeight=metrics.heightPixels;
        WallpaperColors colors=manager.getWallpaperColors(WallpaperManager.FLAG_SYSTEM);
        if(colors!=null)fallback=ContrastMath.colorFor(new double[]{ContrastMath.luminance(colors.getPrimaryColor().toArgb())});
        try{
            if(manager.getWallpaperInfo()!=null)return; // Live wallpapers use their published colors.
            manager.forgetLoadedWallpaper();
            Drawable wallpaper=manager.getDrawable();
            if(wallpaper==null)return;
            int w=wallpaper.getIntrinsicWidth(),h=wallpaper.getIntrinsicHeight();
            if(w<=0||h<=0){w=screenWidth;h=screenHeight;}
            sample=Bitmap.createBitmap(96,Math.max(1,96*screenHeight/screenWidth),Bitmap.Config.ARGB_8888);
            Canvas canvas=new Canvas(sample);
            float scale=Math.max(screenWidth/(float)w,screenHeight/(float)h);
            canvas.scale(sample.getWidth()/(float)screenWidth,sample.getHeight()/(float)screenHeight);
            canvas.translate((screenWidth-w*scale)/2,(screenHeight-h*scale)/2);
            canvas.scale(scale,scale);wallpaper.setBounds(0,0,w,h);wallpaper.draw(canvas);
            Log.i("Kyf37Contrast","Wallpaper sampled: "+sample.getWidth()+"x"+sample.getHeight());
        }catch(RuntimeException e){sample=null;Log.w("Kyf37Contrast","Using wallpaper color fallback",e);}
    }
    public void apply(TextView text){
        if(text==null||text.getWidth()==0||text.getHeight()==0)return;
        int color=fallback;
        if(sample!=null){
            int[] point=new int[2];text.getLocationOnScreen(point);
            float textWidth=Math.min(text.getWidth(),text.getPaint().measureText(text.getText().toString()));
            float textLeft=point[0]+(text.getWidth()-textWidth)/2;
            double[] values=new double[35];int n=0;
            for(int y=0;y<5;y++)for(int x=0;x<7;x++){
                int sx=Math.round((textLeft+textWidth*(x+.5f)/7)*sample.getWidth()/screenWidth);
                int sy=Math.round((point[1]+text.getHeight()*(y+.5f)/5)*sample.getHeight()/screenHeight);
                sx=Math.max(0,Math.min(sample.getWidth()-1,sx));sy=Math.max(0,Math.min(sample.getHeight()-1,sy));
                values[n++]=ContrastMath.luminance(sample.getPixel(sx,sy));
            }
            color=ContrastMath.colorFor(values);
        }
        text.setTextColor(color);
        float d=context.getResources().getDisplayMetrics().density;
        text.setShadowLayer(1.5f*d,0,0,color==ContrastMath.LIGHT?0xDD000000:0xAAFFFFFF);
    }
}
